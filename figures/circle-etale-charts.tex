  \def\R{1.3}
  \tikzset{bandA/.style={draw=figOpenA, line width=7pt, draw opacity=0.5, line cap=butt},
           bandB/.style={draw=figOpenB, line width=7pt, draw opacity=0.5, line cap=butt},

  % ================= (a) dg/dy = 2y != 0, project to x =================
  \begin{scope}
    \draw[bandA] (8:\R) arc[start angle=8, end angle=172, radius=\R];
    \draw[bandA] (188:\R) arc[start angle=188, end angle=352, radius=\R];

    \pgfmathsetmacro{\xa}{0.55}\pgfmathsetmacro{\ya}{sqrt(\R*\R-\xa*\xa)}

    \node[pt] at (\xa,\ya) {}; \node[pt] at (\xa,-\ya) {};
    \node[hollowpt] at (\R,0) {}; \node[hollowpt] at (-\R,0) {};

    % target: the x-axis, image is the open interval (-1,1)
    \draw[axis] (-1.9,-2.3) -- (1.9,-2.3) node[lbl, anchor=west] {$x$};
    \draw[draw=figOpenA!70!black, line width=2pt] (-\R,-2.3) -- (\R,-2.3);
    \node[hollowpt] at (\R,-2.3) {}; \node[hollowpt] at (-\R,-2.3) {};

  \end{scope}
  % ================= (b) dg/dx = 2x != 0, project to y =================
